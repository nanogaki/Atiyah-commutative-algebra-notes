\begin{exercise}[Cohen's theorem]\label{ex7.1}
	Let $\mathfrak{a}$ be a maximal element in $\Sigma$. If $xy\in\mathfrak{a}$ but $x,y\notin\mathfrak{a}$, then $\mathfrak{a}+(x)$ is finitely generated. Let $\left\{x,a_1,\cdots,a_n\right\}$ be its generators and let $\mathfrak{a}_0=\left(a_1,\cdots,a_n\right)$. Prove that $\mathfrak{a}_0\nsubseteq\mathfrak{a}$ is impossible. Therefore, if $u\in\mathfrak{a}\subseteq\mathfrak{a}+(x)=\mathfrak{a}_0+(x)$, $u=a_0+bx$ with $b\in\mathfrak{a}$, so $\mathfrak{a}\subseteq\mathfrak{a}_0+x\cdot(\mathfrak{a}:x)$; we also have $x\cdot(\mathfrak{a}:x)\subseteq\mathfrak{a}$. Note that $y\in(\mathfrak{a}:x)$.
\end{exercise}

\begin{exercise}[\textcolor{red}{Hilbert's Nullstellensatz}]\label{ex7.14}
	If $f\notin\sqrt{\mathfrak{a}}$, there is a $\mathfrak{p}\supseteq\mathfrak{a}$ such that $f\notin\mathfrak{p}$. $(A/\mathfrak{p})_{\overline{f}}$ is a finitely generated $k$-algebra, so by \hyperref[7.9]{Zariski's lemma}, $\left.\left((A/\mathfrak{p})_{\overline{f}}\right)\middle/\mathfrak{m}\right.=k$. Let $x_i$ be the images of $t_i$ in $\left.\left((A/\mathfrak{p})_{\overline{f}}\right)\middle/\mathfrak{m}\right.$, then $x=\left(x_1,\cdots,x_n\right)$ satisfies $x\in V$ but $f(x)\neq0$.
	
	\textcolor{red}{Rabinowitsch trick:} \textcolor{green!70!black}{Let $f_1,\cdots,f_m$ be generators of $\mathfrak{a}$. If $f\in I(V)$, then $f_1,\cdots,f_m,1-t_0f$ has no zeros in $k\left[t_0,t_1,\cdots,t_n\right]$. Apply \Cref{ex5.17}, $1=g_0\left(1-t_0f\right)+\sum_{i=1}^mg_if_i$. Now put $t_0=1/f$, we have $g_i\in k\left[t_1,\cdots,t_n,1/f\right]$. Therefore, $g_i=h_i/f^{N_i}$ with $h_i\in k\left[t_1,\cdots,t_n\right]$. Hence we have $f^N=\sum_{i=1}^mh_if^{N-N_i}f_i$ for all $N\geqslant\max_i\left\{N_i\right\}$.}
\end{exercise}

\begin{exercise}\label{ex7.15}
	\begin{enumerate}
		\item[(iv) $\Rightarrow$ (i)] Let $x_1,\cdots,x_n$ be elements of $M$ whose images in $M/\mathfrak{m}M$ form a basis of $k$-vector space. Let $N=\left(x_1,\cdots,x_n\right)$, so $N+\mathfrak{m}M=M$. Since $\mathfrak{J}=\mathfrak{m}$, apply \hyperref[2.6]{Nakayama's lemma} on $M/N$, we have $M=N$. Let $F$ be a free $A$-module with basis $e_1,\cdots,e_n$, and $\phi:F\rightarrow M$ be $\phi\left(e_i\right)=x_i$. Then 
		\begin{equation*}
			\operatorname{Tor}_1^A(k,M)\rightarrow k\otimes_A\ker\phi\rightarrow k\otimes_AF\xrightarrow{1\otimes\phi}k\otimes_AM\rightarrow0
		\end{equation*}
		exact. Since $k\otimes F$ and $k\otimes M=M/\mathfrak{m}M$ are vector spaces of a same dimension, $1\otimes\phi$ is an isomorphism, so $k\otimes\ker\phi=0$. By \Cref{ex2.3}, $\ker\phi=0$.
	\end{enumerate}
\end{exercise}

\begin{exercise}\label{ex7.19}
	Let $\mathfrak{a}_{kl}=\left(\bigcap_{i\neq k}^r\mathfrak{b}_i\right)\cap\mathfrak{c}_l$, then $\bigcap_{l=1}^s\mathfrak{a}_{kl}=\mathfrak{a}$ for all $k$. Since $\mathfrak{a}\subseteq\mathfrak{b}_k$, the projection $\pi:A\rightarrow A/\mathfrak{b}_k$ maps the equation to $\bigcap_{l=1}^s\pi\left(\mathfrak{a}_{kl}\right)=0$. Since $\mathfrak{b}_k$ irreducible, $\pi\left(\mathfrak{a}_{kl}\right)=0$ for some $l$ and therefore $\mathfrak{a}_{kl}\subseteq\mathfrak{b}_k$. The construction of $\mathfrak{a}_{kl}$ implies $\mathfrak{a}\subseteq\mathfrak{a}_{kl}\subseteq\mathfrak{b}_i$ for all $i\neq k$. Take the intersection of all $i$, we conclude that for any $k$, there is an $l$ such that $\mathfrak{a}_{kl}=\mathfrak{a}$. Therefore, $r=s$, and use \Cref{4.5}.
\end{exercise}

\begin{exercise}\label{ex7.23}
	It is enough to prove $f^*(U\cap C)$ is constructible. Let $C=V(\mathfrak{a})$, by $\operatorname{Spec}(B/\mathfrak{a})\hookrightarrow\operatorname{Spec}(B)\xrightarrow{f^*}\operatorname{Spec}(A)$, we only need to show $f^*(U)$ is constructible. By \Cref{ex6.6} and \Cref{ex1.17.7}, $U=\bigcup_{i=1}^nX_{g_i}$. By $\operatorname{Spec}\left(B_{g_i}\right)\hookrightarrow\operatorname{Spec}(B)\xrightarrow{f^*}\operatorname{Spec}(A)$, it suffices to show $f^*(Y)$ is constructible and take $B=B_{g_i}$. By \Cref{ex6.7}, we can assume $B$ is an integral domain.
	
	Let $X_0$ irreducible and suppose $f^*(Y)\cap X_0$ is dense in $X_0$. By \Cref{ex1.20.4}, $X_0=V(\mathfrak{p})$, so $f^{*-1}\left(X_0\right)=V(f(\mathfrak{p})B)=\operatorname{Spec}\left(A/\mathfrak{p}\otimes_AB\right)$ by \hyperref[ex1.21]{\Cref*{ex1.21}(ii)} and \Cref{ex2.2}. For any $X_0$, we also have $f^*(Y)\cap X_0=f^*\left(f^{*-1}\left(X_0\right)\right)$. Using the universal property of quotient rings to restrict $f$ on $A/\mathfrak{p}$, denoted as $\tilde{f}$, so $\tilde{f}^*(Y)=f^*\left(\operatorname{Spec}\left(A/\mathfrak{p}\otimes_AB\right)\right)=f^*(Y)\cap X_0$ is dense in $X_0$. By \hyperref[ex1.21]{\Cref*{ex1.21}(v)},   $\ker\tilde{f}\subseteq\mathfrak{R}_{A/\mathfrak{p}}=0$, so $\tilde{f}$ is injective. Replacing $A$ with $A/\mathfrak{p}$, we may assume $A$ is an integral domain.
	
	Suppose $A\subseteq B$ are integral domains. By \Cref{ex5.21}, there exists $s\neq0$ such that each $g_\mathfrak{p}:A\rightarrow\operatorname{Frac}(A/\mathfrak{p})\rightarrow\overline{\operatorname{Frac}(A/\mathfrak{p})}$ can be extended to $\tilde{g_\mathfrak{p}}:B\rightarrow\overline{\operatorname{Frac}(A/\mathfrak{p})}$. If $s\notin\mathfrak{p}$, $\mathfrak{q}=\ker\left(\tilde{g_\mathfrak{p}}\right)$ satisfies $\mathfrak{q}\cap A=\mathfrak{p}$, so $X_s\subseteq\tilde{f}^*(Y)$. By \Cref{ex7.21}, $\tilde{f}^*(Y)$ is constructible.
\end{exercise}

\begin{exercise}\label{ex7.24}
	Let $U$ be an open set in $Y$. By \Cref{ex7.23}, $f^*(U)$ is a constructible subset of $X$. Since $f$ has the going-down property, using $\mathfrak{q}\in U$ $\Rightarrow$ $\operatorname{Spec}(B_\mathfrak{q})\subseteq U$, we have $\mathfrak{p}^\prime\in f^*(U)$ for each $\mathfrak{p}^\prime\subseteq\mathfrak{p}\in f^*(U)$.

	Let $X_0$ be an irreducible closed subset of $X$ and suppose $f^*(U)\cap X_0\neq\varnothing$. By \Cref{ex1.20.4}, $X_0=V\left(\mathfrak{p}_0\right)$, so $\mathfrak{p}_0\in f^*(U)$, and therefore $f^*(U)\cap X_0$ is dense in $X_0$. By \Cref{ex7.21.2}, $f^*(U)\cap X_0$ contains a non-empty open subset, and by \Cref{ex7.22}, $f^*(U)$ is open.
\end{exercise}
